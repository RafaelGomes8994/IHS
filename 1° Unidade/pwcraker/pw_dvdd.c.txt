// Não acho que consigo ir mais longe que isso :/
#define DO_ROUND(shift, part) \
    ri1 = _mm256_add_epi32(_mm256_mullo_epi32(ri1, a), c); \
    ri2 = _mm256_add_epi32(_mm256_mullo_epi32(ri2, a), c); \
    folded1 = _mm256_xor_si256(ri1, _mm256_srli_epi32(ri1, 16)); \
    folded2 = _mm256_xor_si256(ri2, _mm256_srli_epi32(ri2, 16)); \
    folded1 = _mm256_xor_si256(folded1, _mm256_srli_epi32(folded1, 8)); \
    folded2 = _mm256_xor_si256(folded2, _mm256_srli_epi32(folded2, 8)); \
    folded1 = _mm256_and_si256(folded1, mask); \
    folded2 = _mm256_and_si256(folded2, mask); \
    if (shift == 0) { \
        acc1_##part = _mm256_xor_si256(acc1_##part, folded1); \
        acc2_##part = _mm256_xor_si256(acc2_##part, folded2); \
    } else { \
        acc1_##part = _mm256_xor_si256(acc1_##part, _mm256_slli_epi32(folded1, shift)); \
        acc2_##part = _mm256_xor_si256(acc2_##part, _mm256_slli_epi32(folded2, shift)); \
    }

    
#define DO_8_ROUNDS() \
    DO_ROUND(0, lo);  DO_ROUND(8, lo);  DO_ROUND(16, lo); DO_ROUND(24, lo); \
    DO_ROUND(0, hi);  DO_ROUND(8, hi);  DO_ROUND(16, hi); DO_ROUND(24, hi); 



typedef struct {

    uint32_t size;
    uint32_t pow;
    uint64_t* keys;
    uint32_t* values;
    
} Hashmap;

static inline uint64_t hashmap_hash (uint64_t key){
    return key * 11400714819323198485ULL;
}

static inline uint32_t hashmap_get (Hashmap* map, uint64_t key){

    uint64_t hashedKey = hashmap_hash(key);
    hashedKey >>= (64 - map->pow);

    uint64_t elementKey;
    uint64_t iteratorKey = hashedKey;
    
    uint32_t wrapAround = (1<<map->pow) - 1;

    while ((elementKey = map->keys[iteratorKey]) != HASHMAP_EMPTY_KEY){
        if (key == elementKey) return map->values[iteratorKey];
        iteratorKey = (iteratorKey + 1) & wrapAround;
    }

    return HASHMAP_NOT_FOUND;

}

static inline void hashmap_insert (Hashmap* map, uint64_t key, uint32_t value){

    uint64_t elementKey;
    uint64_t iteratorKey = hashmap_hash(key) >> (64 - map->pow);
    
    uint32_t wrapAround = (1<<map->pow) - 1;

    while ((elementKey = map->keys[iteratorKey]) != HASHMAP_EMPTY_KEY){
        if (key == elementKey){
            map->values[iteratorKey] = value;
            return;
        }

        iteratorKey = (iteratorKey + 1) & wrapAround;
    }

    map->keys[iteratorKey] = key;
    map->values[iteratorKey] = value;

}

static inline int32_t get_pw_size (uint32_t pw){
    uint32_t count = 0;
    uint8_t* pw_as_uint8 = (uint8_t*) &pw;
    

    count += pw_as_uint8[0] && 1;
    count += pw_as_uint8[1] && 1;
    count += pw_as_uint8[2] && 1;
    count += pw_as_uint8[3] && 1;


    return count;
}

volatile static inline uint32_t generate_next_pw (volatile register uint32_t curr){

    curr++;

    if ((curr & 0x000000FF) == 0x0000003F) curr += (CARRY_OUT_CONSTANT << (8 * 0));
    if ((curr & 0x0000FF00) == 0x00003F00) curr += (CARRY_OUT_CONSTANT << (8 * 1));
    if ((curr & 0x00FF0000) == 0x003F0000) curr += (CARRY_OUT_CONSTANT << (8 * 2));
    if ((curr & 0xFF000000) == 0x3F000000) curr += (CARRY_OUT_CONSTANT << (8 * 3));

    return curr;

}

static inline uint8_t mao_fold32(uint32_t x) {
    x ^= (x >> 16);
    x ^= (x >> 8);
    return (uint8_t)x;
}

static inline uint32_t mao_seed_from_pw(uint32_t pw) {
    uint32_t i;
    uint32_t n = get_pw_size(pw);
    uint32_t seed = 0;
    uint8_t* pw_as_uint8 = (uint8_t*) &pw;

    for (i = 0; i < n; i++) {
        seed = (seed << 8) | ((seed >> 24) ^ val_to_char[pw_as_uint8[n - i - 1]]);
    }

    return seed;
}


static inline void MAO_64(uint8_t* hash, uint32_t pw) {
    uint32_t seed = mao_seed_from_pw(pw);

    // Acumula 8 bytes por bloco em vetor SSE (4 blocos de 8 rodadas = 32 rodadas).
    uint32_t ri = seed;
    uint8_t block_bytes[8];
    __m128i hash_vec = _mm_loadl_epi64((const __m128i*)hash);

    for (uint32_t block = 0; block < 4; block++) {
        for (uint32_t lane = 0; lane < 8; lane++) {
            ri = (1103515245u * ri) + 12345u;
            block_bytes[lane] = mao_fold32(ri);
        }

        __m128i block_vec = _mm_loadl_epi64((const __m128i*)block_bytes);
        hash_vec = _mm_xor_si128(hash_vec, block_vec);
    }

    _mm_storel_epi64((__m128i*)hash, hash_vec);
}

static inline uint64_t hash_bytes_to_uint64(uint8_t hash[8]) {
    // Lê os 8 bytes da memória de uma vez e inverte a ordem em 1 ciclo de clock
    return __builtin_bswap64(*(uint64_t*)hash); 
}

__attribute__((target("avx2")))
static inline void MAO_64_batch16_avx2(uint64_t* out_hashes, const uint32_t* pws, uint32_t count) {
    uint32_t seeds1[8] = {0};
    uint32_t seeds2[8] = {0};

    // Divide as senhas nos dois vetores de sementes
    for (uint32_t lane = 0; lane < count; lane++) {
        if (lane < 8) {
            seeds1[lane] = mao_seed_from_pw(pws[lane]);
        } else {
            seeds2[lane - 8] = mao_seed_from_pw(pws[lane]);
        }
    }

    __m256i ri1 = _mm256_loadu_si256((const __m256i*)seeds1);
    __m256i ri2 = _mm256_loadu_si256((const __m256i*)seeds2);
    
    const __m256i a = _mm256_set1_epi32((int32_t)1103515245u);
    const __m256i c = _mm256_set1_epi32((int32_t)12345u);
    const __m256i mask = _mm256_set1_epi32(0xFF);

    // Utilizamos apenas 4 acumuladores em vez de 16! 
    // lo = armazena os bytes das rodadas 0 a 3. hi = bytes das rodadas 4 a 7.
    __m256i acc1_lo = _mm256_setzero_si256();
    __m256i acc1_hi = _mm256_setzero_si256();
    __m256i acc2_lo = _mm256_setzero_si256();
    __m256i acc2_hi = _mm256_setzero_si256();

    __m256i folded1, folded2;

    
    // Loop Unrolling manual de 32 rodadas
    DO_8_ROUNDS();
    DO_8_ROUNDS();
    DO_8_ROUNDS();
    DO_8_ROUNDS();

    #undef DO_8_ROUNDS
    #undef DO_ROUND

    // Descarregamento alinhado para a memória
    alignas(32) uint32_t raw_acc1_lo[8];
    alignas(32) uint32_t raw_acc1_hi[8];
    alignas(32) uint32_t raw_acc2_lo[8];
    alignas(32) uint32_t raw_acc2_hi[8];

    _mm256_storeu_si256((__m256i*)raw_acc1_lo, acc1_lo);
    _mm256_storeu_si256((__m256i*)raw_acc1_hi, acc1_hi);
    _mm256_storeu_si256((__m256i*)raw_acc2_lo, acc2_lo);
    _mm256_storeu_si256((__m256i*)raw_acc2_hi, acc2_hi);

    // MONTAGEM DO HASH: Sem arrays de char, tudo via registradores ALU.
    // __builtin_bswap32 inverte os 4 bytes perfeitamente para simular o Big Endian nativamente
    for (uint32_t lane = 0; lane < count; lane++) {
        if (lane < 8) {
            uint64_t top = __builtin_bswap32(raw_acc1_lo[lane]);
            uint64_t bot = __builtin_bswap32(raw_acc1_hi[lane]);
            out_hashes[lane] = (top << 32) | bot;
        } else {
            uint64_t top = __builtin_bswap32(raw_acc2_lo[lane - 8]);
            uint64_t bot = __builtin_bswap32(raw_acc2_hi[lane - 8]);
            out_hashes[lane] = (top << 32) | bot;
        }
    }
}



uint32_t answers[1024] = {0};

static inline void MAO_64_batch(uint64_t* out_hashes, const uint32_t* pws, uint32_t count) {
    uint32_t i;

    if (count == 0) {
        return;
    }

#if defined(__x86_64__) || defined(__i386__)
    if (count >= 4 && __builtin_cpu_supports("avx2")) {
        MAO_64_batch16_avx2(out_hashes, pws, count);
        return;
    }
#endif

    for (i = 0; i < count; i++) {
        uint8_t hash_bytes[8] = {0};
        MAO_64(hash_bytes, pws[i]);
        out_hashes[i] = hash_bytes_to_uint64(hash_bytes);
    }
}


typedef struct {
    Hashmap* map;
    uint32_t initial_pw;
    uint32_t last_pw;
} thread_args;

static inline void* BEM_64 (void* ARGS){

    thread_args* args = (thread_args*) ARGS;

    volatile register uint32_t pw = args->initial_pw;
    volatile register uint32_t last_pw = args->last_pw;

    while (pw != last_pw) {
        uint32_t batch_size = 0;
        uint32_t pw_batch[16];
        uint64_t hash_batch[16];

        while (batch_size < 16 && pw != last_pw) {
            pw = generate_next_pw(pw);
            pw_batch[batch_size] = pw;
            batch_size++;
        }

        MAO_64_batch(hash_batch, pw_batch, batch_size);
        
        for (uint32_t i = 0; i < batch_size; i++) {
            uint32_t getPos = hashmap_get(args->map, hash_batch[i]);
            if (getPos != HASHMAP_NOT_FOUND) {
                answers[getPos] = pw_batch[i];
            }
            
        }

    }

    return NULL;

}
